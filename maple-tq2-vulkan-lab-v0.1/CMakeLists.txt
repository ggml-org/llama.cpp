cmake_minimum_required(VERSION 3.20)
project(maple_tq2_vulkan_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MAPLE_BUILD_VULKAN "Build the standalone Vulkan replay harness and shaders" OFF)
add_executable(test-tq2-bits tests/test_bits.cpp)
target_include_directories(test-tq2-bits PRIVATE include)
if(MSVC)
 target_compile_options(test-tq2-bits PRIVATE /W4)
else()
 target_compile_options(test-tq2-bits PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME tq2-bits COMMAND test-tq2-bits)
if(MAPLE_BUILD_VULKAN)
 find_package(Vulkan REQUIRED)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 find_program(GLSLC glslc HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/bin" REQUIRED)
 add_executable(maple-tq2-vk src/vk_replay.cpp)
 target_link_libraries(maple-tq2-vk PRIVATE Vulkan::Vulkan)
 add_custom_target(maple-shaders ALL
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_shaders.py
    --glslc ${GLSLC} --output ${CMAKE_CURRENT_BINARY_DIR}/shaders
  VERBATIM)
 add_dependencies(maple-tq2-vk maple-shaders)
endif()
