cmake_minimum_required(VERSION 3.20)
project(maple_w2_xmx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MAPLE_BUILD_SYCL "Build Intel SYCL/ESIMD W2A16 and native W2A8" OFF)
include(CTest)
add_executable(maple-reference-tests tests/reference_tests.cpp)
target_include_directories(maple-reference-tests PRIVATE include)
add_test(NAME maple-reference-tests COMMAND maple-reference-tests)
add_executable(maple-a8-reference-tests tests/w2a8_reference_tests.cpp)
target_include_directories(maple-a8-reference-tests PRIVATE include)
add_test(NAME maple-a8-reference-tests COMMAND maple-a8-reference-tests)
add_executable(maple-moe-reference-tests tests/moe_reference_tests.cpp)
target_include_directories(maple-moe-reference-tests PRIVATE include)
add_test(NAME maple-moe-reference-tests COMMAND maple-moe-reference-tests)
add_executable(maple-grouping-reference-tests tests/grouping_reference_tests.cpp)
target_include_directories(maple-grouping-reference-tests PRIVATE include)
add_test(NAME maple-grouping-reference-tests COMMAND maple-grouping-reference-tests)
add_executable(maple-architecture-reference-tests tests/architecture_reference_tests.cpp)
target_include_directories(maple-architecture-reference-tests PRIVATE include)
add_test(NAME maple-architecture-reference-tests COMMAND maple-architecture-reference-tests)
# This target uses a NON-EXECUTING SYCL double; tests host DAG wiring only.
add_executable(maple-submission-mock-tests tests/moe_submission_tests.cpp src/maple_moe.cpp src/expert_grouping.cpp)
target_include_directories(maple-submission-mock-tests PRIVATE tests/mock_sycl include)
add_test(NAME maple-submission-mock-tests COMMAND maple-submission-mock-tests)
add_executable(maple-token-reuse-emulation-tests tests/token_reuse_emulation_tests.cpp src/maple_w2a8_grouped.cpp)
target_include_directories(maple-token-reuse-emulation-tests PRIVATE tests/mock_sycl include)
add_test(NAME maple-token-reuse-emulation-tests COMMAND maple-token-reuse-emulation-tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME python-tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
  set_tests_properties(python-tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
if(MAPLE_BUILD_SYCL)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "IntelLLVM|Clang")
    message(FATAL_ERROR "Use oneAPI DPC++ icx/icpx. Ordinary MSVC/g++ cannot compile these SYCL sources.")
  endif()
  add_library(maple-w2-xmx STATIC src/maple_w2a16.cpp src/maple_w2a8.cpp src/maple_moe.cpp src/expert_grouping.cpp src/maple_w2a8_grouped.cpp)
  target_include_directories(maple-w2-xmx PUBLIC include)
  target_compile_options(maple-w2-xmx PUBLIC -fsycl PRIVATE -O3 -fno-fast-math -fsycl-device-code-split=per_kernel)
  target_link_options(maple-w2-xmx PUBLIC -fsycl -fsycl-device-code-split=per_kernel)
  add_executable(maple-w2-compare tools/compare.cpp)
  target_link_libraries(maple-w2-compare PRIVATE maple-w2-xmx)
  target_compile_options(maple-w2-compare PRIVATE -O3 -fno-fast-math)
  add_executable(maple-moe-compare tools/moe_compare.cpp)
  target_link_libraries(maple-moe-compare PRIVATE maple-w2-xmx)
  target_compile_options(maple-moe-compare PRIVATE -O3 -fno-fast-math)
  add_executable(maple-grouping-compare tools/grouping_compare.cpp)
  target_link_libraries(maple-grouping-compare PRIVATE maple-w2-xmx)
  target_compile_options(maple-grouping-compare PRIVATE -O3 -fno-fast-math)
  add_executable(maple-architecture-compare tools/architecture_compare.cpp)
  target_link_libraries(maple-architecture-compare PRIVATE maple-w2-xmx)
  target_compile_options(maple-architecture-compare PRIVATE -O3 -fno-fast-math)
  if(Python3_FOUND)
    add_custom_command(TARGET maple-architecture-compare POST_BUILD
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/stamp_architecture_build.py
        --exe $<TARGET_FILE:maple-architecture-compare>
      COMMENT "Record architecture executable/source identities (not GPU validation)")
  endif()
  # Preserve v0.1 CLI for old test scripts; both executables share one compiled library.
  add_executable(maple-w2-bench tools/bench.cpp)
  target_link_libraries(maple-w2-bench PRIVATE maple-w2-xmx)
  target_compile_options(maple-w2-bench PRIVATE -O3 -fno-fast-math)
endif()
