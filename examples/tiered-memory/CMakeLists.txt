set(TARGET llama-tiered-plan)
add_executable(${TARGET} tiered-plan.cpp)
target_link_libraries(${TARGET} PRIVATE llama-common ggml)
target_compile_features(${TARGET} PRIVATE cxx_std_17)
install(TARGETS ${TARGET} RUNTIME)

if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(llama-tiered-preload SHARED tiered-preload.cpp)
    target_link_libraries(llama-tiered-preload PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    target_compile_features(llama-tiered-preload PRIVATE cxx_std_17)
    set_target_properties(llama-tiered-preload PROPERTIES PREFIX "lib" OUTPUT_NAME "llama-tiered-preload")
    install(TARGETS llama-tiered-preload LIBRARY)
endif()
